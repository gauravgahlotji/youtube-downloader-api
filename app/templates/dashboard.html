app/Services/DownloadEngineService.php 
<?php

namespace App\Services;

use Illuminate\Support\Facades\Http;

class DownloadEngineService
{
    protected string $apiUrl = '{{ base_url }}/api/v1';
    protected string $apiKey = '{{ api_key }}';
    protected string $secretKey = '{{ secret_key }}';

    public function startVideoDownload(string $videoUrl, string $quality = 'best', string $format = 'mp4')
    {
        $endpoint = $this->apiUrl . '/download/video';
        $timestamp = time();
        $nonce = uniqid('n_', true);
        
        $body = json_encode([
            'url' => $videoUrl,
            'quality' => $quality,
            'format' => $format,
            'webhook_url' => route('api.download.webhook')
        ]);

        $signature = hash_hmac('sha256', "{$this->apiKey}.{$timestamp}.{$nonce}.{$body}", $this->secretKey);

        $response = Http::withHeaders([
            'X-API-Key' => $this->apiKey,
            'X-Timestamp' => $timestamp,
            'X-Nonce' => $nonce,
            'X-Signature' => $signature,
            'Content-Type' => 'application/json',
        ])->withBody($body, 'application/json')->post($endpoint);

        return $response->json();
    }

    public function checkJobStatus(string $jobId)
    {
        $endpoint = $this->apiUrl . "/jobs/{$jobId}";
        $timestamp = time();
        $nonce = uniqid('n_', true);
        $signature = hash_hmac('sha256', "{$this->apiKey}.{$timestamp}.{$nonce}.", $this->secretKey);

        return Http::withHeaders([
            'X-API-Key' => $this->apiKey,
            'X-Timestamp' => $timestamp,
            'X-Nonce' => $nonce,
            'X-Signature' => $signature,
        ])->get($endpoint)->json();
    }
}
download.php 
<?php
$apiUrl = '{{ base_url }}/api/v1/download/video';
$apiKey = '{{ api_key }}';
$secretKey = '{{ secret_key }}';
$timestamp = time();
$nonce = bin2hex(random_bytes(8));

$payload = json_encode(['url' => 'https://www.youtube.com/watch?v=dQw4w9WgXcQ', 'quality' => 'best']);
$signature = hash_hmac('sha256', "{$apiKey}.{$timestamp}.{$nonce}.{$payload}", $secretKey);

$ch = curl_init($apiUrl);
curl_setopt_array($ch, [
    CURLOPT_POST => true,
    CURLOPT_POSTFIELDS => $payload,
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER => [
        'Content-Type: application/json',
        "X-API-Key: {$apiKey}",
        "X-Timestamp: {$timestamp}",
        "X-Nonce: {$nonce}",
        "X-Signature: {$signature}"
    ]
]);
$response = curl_exec($ch);
curl_close($ch);
echo $response;
client.py 
import time
import hmac
import hashlib
import json
import requests

api_url = "{{ base_url }}/api/v1/download/video"
api_key = "{{ api_key }}"
secret_key = "{{ secret_key }}"

timestamp = str(int(time.time()))
nonce = str(time.time_ns())
payload = json.dumps({"url": "https://www.youtube.com/watch?v=dQw4w9WgXcQ", "quality": "1080p"})

raw_msg = f"{api_key}.{timestamp}.{nonce}.{payload}".encode('utf-8')
signature = hmac.new(secret_key.encode('utf-8'), raw_msg, hashlib.sha256).hexdigest()

headers = {
    "X-API-Key": api_key,
    "X-Timestamp": timestamp,
    "X-Nonce": nonce,
    "X-Signature": signature,
    "Content-Type": "application/json"
}

res = requests.post(api_url, data=payload, headers=headers)
print(res.json())
client.js 
const crypto = require('crypto');
const axios = require('axios');

const apiUrl = '{{ base_url }}/api/v1/download/video';
const apiKey = '{{ api_key }}';
const secretKey = '{{ secret_key }}';

const timestamp = Math.floor(Date.now() / 1000).toString();
const nonce = crypto.randomBytes(8).toString('hex');
const body = JSON.stringify({ url: 'https://www.youtube.com/watch?v=dQw4w9WgXcQ', quality: '720p' });

const signature = crypto.createHmac('sha256', secretKey)
    .update(`${apiKey}.${timestamp}.${nonce}.${body}`)
    .digest('hex');

axios.post(apiUrl, body, {
    headers: {
        'X-API-Key': apiKey,
        'X-Timestamp': timestamp,
        'X-Nonce': nonce,
        'X-Signature': signature,
        'Content-Type': 'application/json'
    }
}).then(res => console.log(res.data));
